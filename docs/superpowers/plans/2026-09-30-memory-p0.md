# FloodMind 记忆增强 P0 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实施记忆增强 P0：语义事实主动注入（三因子打分）、四类操作固化管线（Graphiti 式失效）、项目级 core memory，三个独立服务 + 薄门面落地。

**Architecture:** 新增 `SemanticMemoryStore`（LTMEntryV2 失效语义）+ `MemoryRecallService` + `MemoryConsolidationService` + `CoreMemoryStore`，由 `MemoryServices` 门面懒加载单例装配；接线点 = native agent 的 memory_messages 构建处 + spawn_background_review 调用处。不修改 DualMemory/ContextRuntime 内部逻辑。

**Tech Stack:** Python 3.13、pydantic v2（BaseModel/model_dump）、pytest、现有 build_agent_tool 工具模式、settings `_cfg` 配置模式。

**Spec:** `docs/superpowers/specs/2026-09-30-memory-p0-design.md`

## Global Constraints

- **禁止打补丁**：不改 `DualMemory`/`LongTermMemory`/`ContextRuntime` 内部逻辑；ContextRuntime 仅允许新增枚举值；旧 store 冻结只读（两个写入方全部重定向后自然冻结）
- **Journal 唯一事实源**：新存储为独立 durable 文件，不动任何 projection
- **原子落盘**：tmp + fsync + os.replace + 进程 RLock（沿用 LongTermMemory 模式）
- **失效语义**：UPDATE = 新条目 ADD + 旧条目打 `invalid_at`/`superseded_by`；DELETE = 软删 `status=deleted`；永不覆盖内容
- **失败回退**：consolidation 任何判定/解析失败 → 本轮全部候选丢弃不写入，绝不 fallback 无条件 ADD
- **配置模式**：`_cfg(cfg, "dotted.key", "ENV_VAR", default)`，env 优先；bool 用 `str(v).lower() in ("1","true","yes","on")`
- **注入顺序**：memory_messages 内 core → experience → recall → history（对应 ContextPriority 15/25/30）
- 提交信息：中文 conventional commits + `Co-Authored-By: Claude Code <noreply@anthropic.com>`
- 测试运行：`python -m pytest <file> -v`（Windows GBK 控制台若炸，加 `PYTHONIOENCODING=utf-8`）

```python
"""语义事实存储（P0 记忆增强新条目模型）。

Graphiti 式失效语义：UPDATE 不覆盖内容，旧条目打 invalid_at/superseded_by，
保留时点回溯（水利参数/方案版本演变场景）。DELETE 为软删除。

数据文件 data/memory/semantic_memory.json，与旧 long_term_memory.json 分离；
门面初始化时一次性导入旧条目（import_legacy，幂等），此后旧 store 冻结只读。
"""

import json
import logging
import os
import re
import threading
import uuid
from datetime import datetime
from typing import List, Optional, Set

from pydantic import BaseModel

logger = logging.getLogger(__name__)

try:
    from floodmind.agent.runtime.services._runtime_root import PROJECT_ROOT as _RUNTIME_ROOT
except Exception:  # pragma: no cover - 防御性回退
    _RUNTIME_ROOT = None

if _RUNTIME_ROOT is not None:
    _DEFAULT_STORE_FILE = str(_RUNTIME_ROOT / "data" / "memory" / "semantic_memory.json")
else:  # pragma: no cover
    _DEFAULT_STORE_FILE = os.path.join(os.path.expanduser("~"), ".floodmind", "semantic_memory.json")

TERM_RE = re.compile(r"[一-龥A-Za-z0-9]{2,}")


def extract_terms(text: str) -> Set[str]:
    """抽取检索词（与 LongTermMemory.search 同口径，抽为纯函数共享）。"""
    return set(TERM_RE.findall(str(text).lower()))


class LTMEntryV2(BaseModel):
    entry_id: str
    content: str
    category: str = "note"
    importance: float = 0.5
    valid_at: str = ""                       # ISO datetime
    invalid_at: Optional[str] = None
    superseded_by: Optional[str] = None      # 指向新条目 entry_id
    status: str = "active"                   # active | superseded | deleted
    access_count: int = 0


class SemanticMemoryStore:
    """语义事实存储：原子落盘 + 进程锁，单进程假设（与 LongTermMemory 一致）。"""

    def __init__(self, store_file: Optional[str] = None):
        self.store_file = store_file or _DEFAULT_STORE_FILE
        self._entries: List[LTMEntryV2] = []
        self._lock = threading.RLock()
        self._load()

    # ── 读 ──────────────────────────────────────────────

    def get(self, entry_id: str) -> Optional[LTMEntryV2]:
        with self._lock:
            for e in self._entries:
                if e.entry_id == entry_id:
                    return e
        return None

    def active_entries(self) -> List[LTMEntryV2]:
        with self._lock:
            return [e for e in self._entries if e.status == "active"]

    def count_active(self) -> int:
        with self._lock:
            return sum(1 for e in self._entries if e.status == "active")

    def find_candidates(self, query_terms: Set[str], top_k: int = 3) -> List[LTMEntryV2]:
        """按关键词重叠取冲突候选（仅 active），供 consolidation LLM 判定前过滤。"""
        if not query_terms:
            return []
        scored = []
        for e in self.active_entries():
            overlap = len(query_terms & extract_terms(e.content))
            if overlap > 0:
                scored.append((overlap, e))
        scored.sort(key=lambda x: x[0], reverse=True)
        return [e for _, e in scored[:top_k]]

    # ── 写 ──────────────────────────────────────────────

    def add(self, content: str, category: str = "note", importance: float = 0.7) -> LTMEntryV2:
        entry = LTMEntryV2(
            entry_id=f"mem_{uuid.uuid4().hex[:8]}",
            content=str(content).strip(),
            category=category or "note",
            importance=float(importance),
            valid_at=datetime.now().isoformat(),
        )
        with self._lock:
            self._entries.append(entry)
            self._save()
        return entry

    def mark_superseded(self, old_id: str, new_id: str) -> bool:
        """UPDATE 的失效语义：旧条目保留内容，打 invalid_at + superseded_by。"""
        with self._lock:
            old = self.get(old_id)
            if old is None or old.status != "active":
                return False
            old.status = "superseded"
            old.invalid_at = datetime.now().isoformat()
            old.superseded_by = new_id
            self._save()
        return True

    def mark_deleted(self, entry_id: str) -> bool:
        """软删除：检索默认过滤，内容保留可回滚。"""
        with self._lock:
            e = self.get(entry_id)
            if e is None or e.status != "active":
                return False
            e.status = "deleted"
            self._save()
        return True

    # ── 持久化 ──────────────────────────────────────────

    def _load(self):
        with self._lock:
            if not os.path.exists(self.store_file):
                self._entries = []
                return
            try:
                with open(self.store_file, "r", encoding="utf-8") as f:
                    raw = json.load(f)
                items = raw.get("entries", []) if isinstance(raw, dict) else raw
                self._entries = [LTMEntryV2(**e) for e in items if isinstance(e, dict)]
            except Exception as e:
                logger.error("[语义记忆] 加载失败: %s", e)
                self._entries = []

    def _save(self):
        try:
            os.makedirs(os.path.dirname(self.store_file), exist_ok=True)
            tmp_path = self.store_file + ".tmp"
            with open(tmp_path, "w", encoding="utf-8") as f:
                json.dump(
                    {"version": 2, "entries": [e.model_dump() for e in self._entries]},
                    f, ensure_ascii=False, indent=2,
                )
                f.flush()
                os.fsync(f.fileno())
            os.replace(tmp_path, self.store_file)
        except Exception as e:
            logger.error("[语义记忆] 保存失败: %s", e)

    def import_legacy(self, legacy_file: Optional[str] = None) -> int:
        """从旧 long_term_memory.json 导入（幂等：按 content 去重），返回新增条数。"""
        if legacy_file is None:
            try:
                from floodmind.memory.dual_memory import LONG_TERM_MEMORY_FILE as legacy_file
            except Exception as e:
                logger.warning("[语义记忆] 旧存储路径解析失败: %s", e)
                return 0
        if not os.path.exists(legacy_file):
            return 0
        try:
            with open(legacy_file, "r", encoding="utf-8") as f:
                items = json.load(f)
        except Exception as e:
            logger.warning("[语义记忆] 旧存储读取失败: %s", e)
            return 0
        if not isinstance(items, list):
            return 0
        with self._lock:
            existing = {e.content for e in self._entries}
            added = 0
            for item in items:
                if not isinstance(item, dict):
                    continue
                content = str(item.get("content", "")).strip()
                if not content or content in existing:
                    continue
                self._entries.append(LTMEntryV2(
                    entry_id=f"mem_{uuid.uuid4().hex[:8]}",
                    content=content,
                    category=str(item.get("category", "note")),
                    importance=float(item.get("importance", 0.5)),
                    valid_at=str(item.get("timestamp", "")) or datetime.now().isoformat(),
                    access_count=int(item.get("access_count", 0)),
                ))
                existing.add(content)
                added += 1
            if added:
                self._save()
        logger.info("[语义记忆] 旧存储导入 %d 条", added)
        return added
```

from datetime import datetime, timedelta

import pytest

from floodmind.memory.recall import MemoryRecallService
from floodmind.memory.semantic_store import SemanticMemoryStore, extract_terms


@pytest.fixture
def store(tmp_path):

def make_service(store, **kw):

    params.update(kw)
    return MemoryRecallService(store, **params)


def test_score_zero_on_disjoint_terms(store):
    e = store.add("报表导出格式要求", category="rule")
    terms = extract_terms("敖江洪水预报")
    assert MemoryRecallService.score(e, terms, datetime.now(), 30.0) == 0.0


def test_score_monotonic_decay(store):
    ts = datetime(2026, 1, 1, 12, 0, 0)
    e = store.add("敖江洪水预报参数", category="parameter", importance=1.0)
    e.valid_at = ts.isoformat()
    terms = extract_terms("敖江预报")
    s1 = MemoryRecallService.score(e, terms, ts, 30.0)
    s2 = MemoryRecallService.score(e, terms, ts + timedelta(days=30), 30.0)
    assert s1 == pytest.approx(1.0 * 1.0 * 1.0)          # relevance=1, recency=1, importance=1
    assert s2 == pytest.approx(math.exp(-1.0))           # 30 天后衰减一个尺度
    assert s2 < s1


def test_score_partial_relevance(store):
    e = store.add("敖江洪水预报", category="note", importance=1.0)
    terms = {"敖江", "预报", "完全无关词"}
    s = MemoryRecallService.score(e, terms, datetime.now(), 30.0)
    assert s == pytest.approx(2 / 3)  # relevance=2/3, recency=1, importance=1

```python
"""语义事实主动注入（P0-1）。

三因子打分（Generative Agents 风格）：score = relevance × recency × importance
- relevance：查询词与条目内容的关键词重叠率
- recency：MemoryBank 艾宾浩斯衰减 exp(-Δt / S)，S=decay_scale_days
- importance：条目 importance 字段（clamp 到 [0,1]）

LTM 百级条目 + 正则打分为毫秒级，每轮装配 system prompt 时重算，不做缓存
（P1 换 embedding 后再加 TTL 缓存）。
"""

import logging
import math
from datetime import datetime
from typing import List, Optional, Set, Tuple

from floodmind.memory.semantic_store import LTMEntryV2, SemanticMemoryStore, extract_terms

logger = logging.getLogger(__name__)


class MemoryRecallService:
    def __init__(
        self,
        store: SemanticMemoryStore,
        *,
        top_k: int = 5,
        min_score: float = 0.05,
        decay_scale_days: float = 30.0,
        max_chars: int = 1500,
    ):
        self._store = store
        self._top_k = top_k
        self._min_score = min_score
        self._decay_scale_days = max(float(decay_scale_days), 1e-6)
        self._max_chars = max_chars

    @staticmethod
    def score(
        entry: LTMEntryV2,
        query_terms: Set[str],
        now: datetime,
        decay_scale_days: float,
    ) -> float:
        if not query_terms:
            return 0.0
        entry_terms = extract_terms(entry.content)
        relevance = len(query_terms & entry_terms) / len(query_terms)
        if relevance <= 0:
            return 0.0
        try:
            valid_at = datetime.fromisoformat(entry.valid_at)
            delta_days = max((now - valid_at).total_seconds(), 0.0) / 86400.0
        except (ValueError, TypeError):
            delta_days = 0.0  # 时间缺失按最新处理，不打断召回
        recency = math.exp(-delta_days / max(float(decay_scale_days), 1e-6))
        importance = min(max(float(entry.importance), 0.0), 1.0)
        return relevance * recency * importance

    def build_block(self, user_input: str, now: Optional[datetime] = None) -> str:
        """按当前输入对 active 条目打分取 top-k，渲染为 ContextBlock 文本；无命中返回 ""。"""
        if not user_input or self._store.count_active() == 0:
            return ""
        now = now or datetime.now()
        terms = extract_terms(user_input)
        if not terms:
            return ""
        scored: List[Tuple[float, LTMEntryV2]] = [
            (self.score(e, terms, now, self._decay_scale_days), e)
            for e in self._store.active_entries()
        ]
        scored = [(s, e) for s, e in scored if s >= self._min_score]
        if not scored:
            return ""
        scored.sort(key=lambda x: x[0], reverse=True)

        lines = ["[长期记忆] 与当前请求可能相关的历史事实（已按相关度排序）："]
        total = len(lines[0])
        for s, e in scored[: self._top_k]:
            item = f"- ({e.category}) {e.content}"
            if total + len(item) > self._max_chars:
                break
            lines.append(item)
            total += len(item) + 1
        if len(lines) == 1:
            return ""
        return "\n".join(lines)
```

```python
"""项目级 durable core memory（P0-3，Letta 原则：小而常驻、可工具自编辑）。

从 per-session core_memory.json 升格为项目级存储，启动时聚合迁移旧会话文件
（按 (category, fact) 去重，只读旧文件）。schema 预留 scope 字段
（user/project/session 三层为接口预留，本次只写 project）。
"""

import json
import logging
import os
import threading
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional

logger = logging.getLogger(__name__)

try:
    from floodmind.agent.runtime.services._runtime_root import PROJECT_ROOT as _RUNTIME_ROOT
except Exception:  # pragma: no cover
    _RUNTIME_ROOT = None

if _RUNTIME_ROOT is not None:
    _DEFAULT_STORE_FILE = str(_RUNTIME_ROOT / "data" / "memory" / "core_memory.json")
else:  # pragma: no cover
    _DEFAULT_STORE_FILE = os.path.join(os.path.expanduser("~"), ".floodmind", "core_memory.json")


def _legacy_session_dir() -> Path:
    if _RUNTIME_ROOT is not None:
        return Path(_RUNTIME_ROOT) / "data" / "sessions"
    return Path(os.path.expanduser("~")) / ".floodmind" / "sessions"


class CoreMemoryStore:
    """项目级核心记忆存储。内存索引 + 原子落盘 + 进程锁。"""

    def __init__(
        self,
        store_file: Optional[str] = None,
        *,
        migrate_legacy: bool = True,
        max_chars: int = 800,
    ):
        self.store_file = store_file or _DEFAULT_STORE_FILE
        self._max_chars = max_chars
        self._facts: Dict[str, List[Dict[str, str]]] = {}
        self._lock = threading.RLock()
        self._load()
        if migrate_legacy:
            self._migrate_legacy_sessions()

    # ── 读 ──────────────────────────────────────────────

    def load(self) -> Dict[str, Any]:
        with self._lock:
            return {
                "updated_at": datetime.now(timezone.utc).isoformat(),
                "facts": {cat: [dict(item) for item in items]
                          for cat, items in self._facts.items()},
            }

    def render(self, *, max_chars: Optional[int] = None) -> str:
        """渲染为注入块的文本；空返回 ""。"""
        limit = max_chars if max_chars is not None else self._max_chars
        with self._lock:
            if not any(self._facts.values()):
                return ""
            header = "[核心记忆] 本项目持久关键事实（可经 CoreMemoryAppend 维护）："
            lines = [header]
            total = len(header)
            for cat, items in self._facts.items():
                if not items:
                    continue
                section = f"## {cat}"
                if total + len(section) > limit:
                    break
                lines.append(section)
                total += len(section) + 1
                for item in items:
                    line = f"- {item['fact']}"
                    if total + len(line) > limit:
                        return "\n".join(lines)
                    lines.append(line)
                    total += len(line) + 1
            return "\n".join(lines)

    # ── 写 ──────────────────────────────────────────────

    def append(self, category: str, fact: str, *, scope: str = "project") -> bool:
        category = str(category).strip()
        fact = str(fact).strip()
        if not category or not fact:
            return False
        with self._lock:
            items = self._facts.setdefault(category, [])
            if any(item["fact"] == fact for item in items):
                return False
            items.append({"fact": fact, "scope": scope or "project"})
            self._save()
        return True

    # ── 持久化 ──────────────────────────────────────────

    def _load(self):
        with self._lock:
            if not os.path.exists(self.store_file):
                self._facts = {}
                return
            try:
                raw = json.loads(Path(self.store_file).read_text(encoding="utf-8"))
                self._facts = self._normalize(raw.get("facts", {}))
            except Exception as e:
                logger.error("[核心记忆] 加载失败: %s", e)
                self._facts = {}

    def _save(self):
        try:
            os.makedirs(os.path.dirname(self.store_file), exist_ok=True)
            tmp_path = self.store_file + ".tmp"
            with open(tmp_path, "w", encoding="utf-8") as f:
                json.dump(
                    {"version": 2, "facts": self._facts},
                    f, ensure_ascii=False, indent=2,
                )
                f.flush()
                os.fsync(f.fsync if False else f.fileno())
            os.replace(tmp_path, self.store_file)
        except Exception as e:
            logger.error("[核心记忆] 保存失败: %s", e)

    # ── 旧会话迁移 ──────────────────────────────────────

    def _migrate_legacy_sessions(self) -> int:
        """扫描 data/sessions/*/core_memory.json，只读聚合导入（(category, fact) 去重）。"""
        sessions_dir = _legacy_session_dir()
        if not sessions_dir.is_dir():
            return 0
        added = 0
        try:
            entries = []  # (category, fact)
            for child in sorted(sessions_dir.iterdir()):
                legacy = child / "core_memory.json"
                if not legacy.is_file():
                    continue
                try:
                    raw = json.loads(legacy.read_text(encoding="utf-8"))
                except Exception as e:
                    logger.warning("[核心记忆] 旧会话文件读取失败 %s: %s", legacy, e)
                    continue
                for cat, facts in (raw.get("facts") or {}).items():
                    for fact in facts or []:
                        if isinstance(fact, str) and fact.strip():
                            entries.append((str(cat), fact.strip()))
            with self._lock:
                for cat, fact in entries:
                    items = self._facts.setdefault(cat, [])
                    if any(item["fact"] == fact for item in items):
                        continue
                    items.append({"fact": fact, "scope": "project"})
                    added += 1
                if added:
                    self._save()
        except Exception as e:
            logger.warning("[核心记忆] 旧会话迁移失败（非致命）: %s", e)
        if added:
            logger.info("[核心记忆] 已从旧会话文件导入 %d 条", added)
        return added

    @staticmethod
    def _normalize(facts: Dict[str, Any]) -> Dict[str, List[Dict[str, str]]]:
        """兼容两种落盘形态：纯字符串列表（旧）或 dict 列表（新）。"""
        out: Dict[str, List[Dict[str, str]]] = {}
        for cat, items in (facts or {}).items():
            norm: List[Dict[str, str]] = []
            for item in items or []:
                if isinstance(item, str) and item.strip():
                    norm.append({"fact": item.strip(), "scope": "project"})
                elif isinstance(item, dict) and str(item.get("fact", "")).strip():
                    norm.append({
                        "fact": str(item["fact"]).strip(),
                        "scope": str(item.get("scope", "project")),
                    })
            out[str(cat)] = norm
        return out
```

"""四类操作固化管线（P0-2，Mem0 式抽取 + Graphiti 式失效）。

BackgroundReview 每轮抽取候选（偏好 + 事实），本服务做单次 LLM 批量判定：
ADD / UPDATE / DELETE / NOOP。UPDATE 失效不覆盖、DELETE 软删，均保留回溯链。

铁律：判定失败 / 解析失败 / 非法 action / 目标不存在 → 本轮全部候选丢弃不写入，
绝不 fallback 成无条件 ADD（那会加剧新旧事实并存，正是本增强要解决的问题）。
"""

import json
import logging
from typing import Any, Dict, List, Optional

from floodmind.memory.semantic_store import SemanticMemoryStore, extract_terms

logger = logging.getLogger(__name__)

_VALID_ACTIONS = {"ADD", "UPDATE", "DELETE", "NOOP"}

_JUDGE_PROMPT = """你是记忆管理助手。对每条候选记忆，决定如何写入长期记忆存储。

## 现有记忆（每条候选的冲突候选）
{existing}

## 候选记忆
{candidates}

## 规则
- ADD：候选是全新事实，与现有记忆无冲突
- UPDATE：候选与某条现有记忆冲突或演化（参数变更、方案替换），target_id 指向被替代的旧条目
- DELETE：某条现有记忆已明确失效或被否定，target_id 指向它
- NOOP：候选是临时信息、重复内容、或无需持久化
- 没有冲突时一律 ADD，不要臆造 UPDATE

## 输出格式
严格输出 JSON 数组（不要其他文字），每个元素：
{{"index": 0, "action": "ADD|UPDATE|DELETE|NOOP", "target_id": "...", "reason": "..."}}
index 为候选序号；ADD/NOOP 不需要 target_id。数组长度必须等于候选数。"""


class MemoryConsolidationService:
    def __init__(
        self,
        store: SemanticMemoryStore,
        model_client: Any = None,
        *,
        conflict_candidate_top_k: int = 3,
    ):
        self._store = store
        self._model_client = model_client
        self._conflict_candidate_top_k = conflict_candidate_top_k

    def set_model_client(self, model_client: Any) -> None:
        self._model_client = model_client

    def consolidate(self, candidates: List[Dict[str, Any]]) -> Dict[str, int]:
        stats = {"added": 0, "updated": 0, "deleted": 0, "noop": 0, "discarded": 0}
        candidates = [c for c in candidates if str(c.get("content", "")).strip()]
        if not candidates:
            return stats

        # 阶段一：判定 + 验证（全部通过才进入阶段二）
        try:
            decisions = self._judge(candidates)
            self._validate(decisions, candidates)
        except Exception as e:
            logger.warning("[记忆固化] 判定失败，本轮 %d 条候选全部丢弃: %s", len(candidates), e)
            stats["discarded"] = len(candidates)
            return stats

        # 阶段二：统一应用
        for d in decisions:
            action = d["action"]
            if action == "ADD":
                self._store.add(candidates[d["index"]]["content"],
                                category=candidates[d["index"]].get("type", "note"))
                stats["added"] += 1
            elif action == "UPDATE":
                new_entry = self._store.add(candidates[d["index"]]["content"],
                                            category=candidates[d["index"]].get("type", "note"))
                if self._store.mark_superseded(d["target_id"], new_entry.entry_id):
                    stats["updated"] += 1
            elif action == "DELETE":
                if self._store.mark_deleted(d["target_id"]):
                    stats["deleted"] += 1
            else:  # NOOP
                stats["noop"] += 1
        logger.info("[记忆固化] added=%d updated=%d deleted=%d noop=%d",
                    stats["added"], stats["updated"], stats["deleted"], stats["noop"])
        return stats

    # ── 内部 ────────────────────────────────────────────

    def _judge(self, candidates: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
        if self._model_client is None:
            raise RuntimeError("model_client 未注入")
        existing_blocks = []
        for i, c in enumerate(candidates):
            terms = extract_terms(c["content"])
            conflicts = self._store.find_candidates(terms, top_k=self._conflict_candidate_top_k)
            conflict_lines = "\n".join(
                f"  - id={e.entry_id} [{e.category}] {e.content}" for e in conflicts
            ) or "  （无）"
            existing_blocks.append(f"候选{i} 的冲突候选：\n{conflict_lines}")
        candidate_lines = "\n".join(
            f"候选{i}: [{c.get('type', 'note')}] {c['content']}" for i, c in enumerate(candidates)
        )
        prompt = _JUDGE_PROMPT.format(
            existing="\n\n".join(existing_blocks), candidates=candidate_lines
        )
        response = self._model_client.invoke(
            prompt=prompt,
            system_prompt="你是记忆管理助手，只输出规定的 JSON 数组。",
            temperature=0.2,
            max_tokens=1500,
        )
        raw = response.content if hasattr(response, "content") else str(response)
        text = raw.strip()
        if text.startswith("```"):
            text = text.strip("`")
            if text.startswith("json"):
                text = text[4:]
        decisions = json.loads(text)
        if not isinstance(decisions, list):
            raise ValueError("判定结果不是数组")
        return decisions

    @staticmethod
    def _validate(decisions: List[Dict[str, Any]], candidates: List[Dict[str, Any]]) -> None:
        if len(decisions) != len(candidates):
            raise ValueError(f"判定数量 {len(decisions)} != 候选数 {len(candidates)}")
        for d in decisions:
            if not isinstance(d, dict):
                raise ValueError("判定元素不是对象")
            if d.get("action") not in _VALID_ACTIONS:
                raise ValueError(f"非法 action: {d.get('action')}")
            idx = d.get("index")
            if not isinstance(idx, int) or not (0 <= idx < len(candidates)):
                raise ValueError(f"非法 index: {idx}")
        # target 存在性校验放在应用前统一做（需要 store）
```

```python
"""记忆服务门面（P0）：装配与生命周期的唯一入口。

三服务（recall/consolidation/core）由本门面懒加载单例装配；
dsh 迁移时门面是唯一重绑点。on_run_completed / on_checkpoint 为
P1/P2 预留钩子，本次空实现。
"""

import logging
import threading
from typing import Any, Optional

from floodmind.config.settings import settings

logger = logging.getLogger(__name__)


class MemoryServices:
    def __init__(self):
        from floodmind.memory.consolidation import MemoryConsolidationService
        from floodmind.memory.core_memory import CoreMemoryStore
        from floodmind.memory.recall import MemoryRecallService
        from floodmind.memory.semantic_store import SemanticMemoryStore

        self.store = SemanticMemoryStore()
        # 一次性导入旧长期记忆（幂等），此后旧 store 冻结只读
        try:
            self.store.import_legacy()
        except Exception as e:
            logger.warning("[记忆门面] 旧长期记忆导入失败（非致命）: %s", e)

        rc = settings.memory_recall
        self.recall: Optional[MemoryRecallService] = (
            MemoryRecallService(
                self.store,
                top_k=rc.top_k,
                min_score=rc.min_score,
                decay_scale_days=rc.decay_scale_days,
                max_chars=rc.max_chars,
            ) if rc.enabled else None
        )

        kc = settings.memory_consolidation
        self.consolidation: Optional[MemoryConsolidationService] = (
            MemoryConsolidationService(
                self.store,
                conflict_candidate_top_k=kc.conflict_candidate_top_k,
            ) if kc.enabled else None
        )

        cc = settings.core_memory
        self.core: Optional[CoreMemoryStore] = (
            CoreMemoryStore(migrate_legacy=cc.migrate_legacy, max_chars=cc.max_chars)
            if cc.enabled else None
        )

    def on_turn_completed(self, review_result: Any) -> None:
        """每轮回顾完成后：偏好+facts 交给四类操作管线判定。"""
        if self.consolidation is None or review_result is None:
            return
        candidates = [
            {"content": p.get("content", ""), "type": p.get("type", "preference")}
            for p in getattr(review_result, "user_preferences", [])
        ]
        candidates += [
            {"content": f.get("content", ""), "type": f.get("type", "fact")}
            for f in getattr(review_result, "facts", [])
        ]
        if candidates:
            self.consolidation.consolidate(candidates)

    def on_run_completed(self, run_summary: Any = None) -> None:
        """P1 会话结束沉淀预留——本次空实现。"""
        return None

    def on_checkpoint(self, checkpoint_ref: Any = None) -> None:
        """P2 记忆演化维护预留——本次空实现。"""
        return None


_instance: Optional[MemoryServices] = None
_lock = threading.Lock()


def get_memory_services() -> MemoryServices:
    global _instance
    with _lock:
        if _instance is None:
            _instance = MemoryServices()
        return _instance
```
